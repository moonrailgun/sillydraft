# 人设（Persona）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 作品通过 `personaId` 绑定全局人设，`{{user}}` / `{{persona}}` 只从人设取值；人设可带外貌、服饰、参考图，并有看图反推、AI 补全润色、生成参考图三个辅助。

**Architecture:** 人设存在全局设置 `personas`，核心包提供模型、仓库扩展（增删改查、引用检查、旧数据迁移）和 `PersonaService`（模型辅助）。显示、生成、新建小说都经 `personaFor(profile)` 解析。项目导出把绑定换成人设快照，恢复时再映射回本机人设。界面新增人设列表、编辑页和选择器。

**Tech Stack:** Dart 3.13.5 / Flutter 3.47.7，drift（SQLite），Riverpod，forui，file_selector，package:image。

**Spec:** `docs/superpowers/specs/2026-10-14-persona-design.md`

```dart
import 'asset.dart';

/// Who `{{user}}` is in a story: [name] fills `{{user}}` and [prompt] fills
/// `{{persona}}`. Appearance and outfit are kept apart so image tools can
/// use them on their own.
final class Persona {
  const Persona({
    required this.id,
    required this.name,
    this.description = '',
    this.appearance = '',
    this.outfit = '',
    this.images = const [],
  });

  factory Persona.fromJson(Map<String, dynamic> json) => Persona(
    id: json['id'] as String,
    name: json['name'] as String,
    description: json['description'] as String? ?? '',
    appearance: json['appearance'] as String? ?? '',
    outfit: json['outfit'] as String? ?? '',
    images: [
      for (final image in json['images'] as List? ?? const [])
        PersonaImage.fromJson(image as Map<String, dynamic>),
    ],
  );

  /// The built-in persona: the reader as “我”, so stories address them in
  /// the first person.
  static const meId = 'me';
  static const me = Persona(id: meId, name: '我');
  static const maxImages = 8;

  final String id, name, description, appearance, outfit;
  final List<PersonaImage> images;

  /// `{{persona}}`: the description, then appearance and outfit lines, each
  /// left out when empty.
  String get prompt => [
    description.trim(),
    if (appearance.trim().isNotEmpty) '外貌：${appearance.trim()}',
    if (outfit.trim().isNotEmpty) '服饰：${outfit.trim()}',
  ].where((s) => s.isNotEmpty).join('\n');

  /// The text fields: what a project export carries, and what makes two
  /// personas the same one.
  Map<String, String> get snapshot => {
    'name': name,
    'description': description,
    'appearance': appearance,
    'outfit': outfit,
  };

  Persona copyWith({
    String? id,
    String? name,
    String? description,
    String? appearance,
    String? outfit,
    List<PersonaImage>? images,
  }) => Persona(
    id: id ?? this.id,
    name: name ?? this.name,
    description: description ?? this.description,
    appearance: appearance ?? this.appearance,
    outfit: outfit ?? this.outfit,
    images: images ?? this.images,
  );

  Map<String, dynamic> toJson() => {
    'id': id,
    ...snapshot,
    'images': [for (final image in images) image.toJson()],
  };
}

/// A reference image, stored by content hash like story assets.
final class PersonaImage {
  const PersonaImage({
    required this.hash,
    required this.thumbnailHash,
    required this.width,
    required this.height,
  });

  factory PersonaImage.of(PreparedAsset prepared) => PersonaImage(
    hash: prepared.hash,
    thumbnailHash: prepared.thumbnailHash,
    width: prepared.width,
    height: prepared.height,
  );

  factory PersonaImage.fromJson(Map<String, dynamic> json) => PersonaImage(
    hash: json['hash'] as String,
    thumbnailHash: json['thumbnailHash'] as String,
    width: json['width'] as int,
    height: json['height'] as int,
  );

  final String hash, thumbnailHash;
  final int width, height;

  Map<String, dynamic> toJson() => {
    'hash': hash,
    'thumbnailHash': thumbnailHash,
    'width': width,
    'height': height,
  };
}
```

```dart
part of 'story_repository.dart';

extension PersonaOperations on StoryRepository {
  /// Every persona, the built-in “我” first even before it is first saved.
  Future<List<Persona>> listPersonas() async {
    final stored = [
      for (final item
          in (await getSetting('personas'))?['items'] as List? ?? const [])
        Persona.fromJson(item as Map<String, dynamic>),
    ];
    return [
      stored.where((p) => p.id == Persona.meId).firstOrNull ?? Persona.me,
      ...stored.where((p) => p.id != Persona.meId),
    ];
  }

  /// The persona a writing profile binds: “我” when it binds none, or one
  /// that no longer exists.
  Future<Persona> personaFor(Map<String, dynamic> profile) async {
    final all = await listPersonas();
    return all.where((p) => p.id == profile['personaId']).firstOrNull ??
        all.first;
  }

  /// Adds or replaces [persona]; a new one (empty id) gets its id here.
  Future<Persona> savePersona(Persona persona) => _write(() async {
    if (persona.name.trim().isEmpty) {
      throw const FormatException('请填写人设名字');
    }
    if (persona.images.length > Persona.maxImages) {
      throw const FormatException('最多 8 张参考图');
    }
    final saved = persona.id.isEmpty ? persona.copyWith(id: _id()) : persona;
    final all = await listPersonas();
    final index = all.indexWhere((p) => p.id == saved.id);
    if (index < 0) {
      all.add(saved);
    } else {
      all[index] = saved;
    }
    await _storePersonas(all);
    return saved;
  });

  /// Refused for “我” and for a persona any novel still binds.
  Future<void> deletePersona(String id) => _write(() async {
    if (id == Persona.meId) throw const FormatException('内置人设“我”不能删除');
    final novels = {
      for (final row in await _rows(
        "SELECT s.value_json, b.novel_id FROM app_settings s JOIN branches b ON s.key = 'writing/' || b.id",
      ))
        if ((jsonDecode(row['value_json'] as String) as Map)['personaId'] == id)
          row['novel_id'] as String,
    };
    if (novels.isNotEmpty) {
      throw FormatException('人设被 ${novels.length} 部作品使用，请先在作品中改用其他人设');
    }
    await _storePersonas([
      for (final p in await listPersonas())
        if (p.id != id) p,
    ]);
  });

  /// Binds writing profiles that still carry the free-text `userName` /
  /// `persona` fields, or a `personaSnapshot` from another installation, to
  /// a persona, and drops those fields. Safe to repeat; runs at startup and
  /// after a project is restored.
  Future<void> migrateLegacyPersonas() => _write(() async {
    final all = await listPersonas();
    var created = false;
    for (final row in await _rows(
      "SELECT key,value_json FROM app_settings WHERE key LIKE 'writing/%'",
    )) {
      final value =
          jsonDecode(row['value_json'] as String) as Map<String, dynamic>;
      final snapshot = value.remove('personaSnapshot'),
          userName = value.remove('userName'),
          text = value.remove('persona');
      if (snapshot == null && userName == null && text == null) continue;
      if (value['personaId'] == null) {
        final source = snapshot is Map ? snapshot : const {};
        String field(String key) =>
            source[key] is String ? source[key] as String : '';
        final name = (snapshot == null ? userName as String? ?? '' : field('name'))
                .trim(),
            description = snapshot == null ? text as String? ?? '' : field('description');
        final wanted = Persona(
          id: '',
          name: name.isEmpty ? Persona.me.name : name,
          description: description,
          appearance: field('appearance'),
          outfit: field('outfit'),
        );
        // The old default, “主角” with no description, becomes “我”.
        final persona =
            snapshot == null &&
                (name.isEmpty || name == '主角') &&
                description.trim().isEmpty
            ? all.first
            : all
                      .where(
                        (p) =>
                            jsonEncode(p.snapshot) ==
                            jsonEncode(wanted.snapshot),
                      )
                      .firstOrNull ??
                  wanted.copyWith(id: _id());
        if (!all.contains(persona)) {
          all.add(persona);
          created = true;
        }
        value['personaId'] = persona.id;
      }
      await setSetting(row['key'] as String, value);
    }
    if (created) await _storePersonas(all);
  });

  Future<void> _storePersonas(List<Persona> all) => setSetting('personas', {
    'items': [for (final p in all) p.toJson()],
  });
}
```

```dart
  /// A stored file by content hash, as persona images refer to theirs.
  File hashFile(String hash, {bool thumbnail = false}) =>
      _path(hash, thumbnail ? 'jpg' : 'png');

  Future<Uint8List> readHash(String hash, {bool thumbnail = false}) async {
    final bytes = await hashFile(hash, thumbnail: thumbnail).readAsBytes();
    if (sha256.convert(bytes).toString() != hash) {
      throw const FormatException('图片文件已损坏');
    }
    return bytes;
  }
```

```dart
  Map<String, Object?> _portableSetting(

  ) {
```

```dart
  /// run uses the nodes the author last saved there.
  Future<ModelConnection> withLatestWorkflow(
    ModelConnection connection,
  ) async {
    final workflow = connection.comfyWorkflow;
    if (connection.protocol != ModelProtocol.comfyUI ||
        workflow == null ||
        workflow.filename.isEmpty) {
      return connection;
    }
    final factory = runner.imageProviderFactory;
    if (factory == null) throw StateError('未配置生图执行器');
    final provider = await factory(connection);

      if (provider is! ComfyUIProvider) {

      }
      final nodes = await provider.readWorkflow(workflow.filename);
      return ModelConnection.fromJson({
        ...connection.raw,
        'comfyWorkflow': ComfyWorkflow.fromJson({
          ...workflow.raw,
          'nodes': nodes,
        }).raw,
      });
    } finally {
      provider.close();
```

/// Model help for filling in a persona. Results only fill the form; the
/// author decides what to save.
final class PersonaService {
  const PersonaService({required this.repository, required this.runner});
  final StoryRepository repository;
  final TaskRunner runner;

  /// Appearance and outfit from the first four reference images, and a
  /// description when [draft] has none.
  Future<Persona> describeImages(
    ModelConnection connection,
    Persona draft, {
    GenerationCancellation? cancellation,
  }) async {
    if (draft.images.isEmpty) throw const FormatException('请先添加参考图');
    if (connection.protocol.isLocalCli) {
      throw const FormatException('当前资料整理模型不支持看图，请改用 OpenAI 兼容连接');
    }
    final library =
        runner.assetLibrary ?? (throw StateError('本地图片目录尚未配置'));
    final reply = await _ask(connection, _describe, [
      {'type': 'text', 'text': jsonEncode(draft.snapshot)},
      // ponytail: 480px thumbnails keep requests small; send larger copies
      // if descriptions miss fine detail.
      for (final image in draft.images.take(4))
        {
          'type': 'image_url',
          'image_url': {
            'url':
                'data:image/jpeg;base64,${base64Encode(await library.readHash(image.thumbnailHash, thumbnail: true))}',
          },
        },
    ], cancellation);
    return draft.copyWith(
      appearance: _text(reply, 'appearance'),
      outfit: _text(reply, 'outfit'),
      description:
          draft.description.trim().isEmpty && reply['description'] is String
          ? _text(reply, 'description')
          : null,
    );
  }

  /// Fills and polishes every field from whatever [draft] has; a written
  /// name stays.
  Future<Persona> refine(
    ModelConnection connection,
    Persona draft, {
    GenerationCancellation? cancellation,
  }) async {
    final reply = await _ask(
      connection,
      _refine,
      jsonEncode(draft.snapshot),
      cancellation,
    );
    return draft.copyWith(
      name: draft.name.trim().isEmpty ? _text(reply, 'name') : null,
      description: _text(reply, 'description'),
      appearance: _text(reply, 'appearance'),
      outfit: _text(reply, 'outfit'),
    );
  }

  /// An image prompt for [painter]: Danbooru tags for NovelAI and ComfyUI,
  /// an English sentence for other image models.
  Future<String> imagePrompt(
    ModelConnection connection,
    Persona draft,
    ModelConnection painter, {
    GenerationCancellation? cancellation,
  }) async {
    if (draft.appearance.trim().isEmpty && draft.outfit.trim().isEmpty) {
      throw const FormatException('请先填写外貌或默认服饰');
    }
    final tags =
        painter.protocol == ModelProtocol.novelAI ||
        painter.protocol == ModelProtocol.comfyUI;
    final reply = await _ask(
      connection,
      tags ? _tags : _sentence,
      jsonEncode({'appearance': draft.appearance, 'outfit': draft.outfit}),
      cancellation,
    );
    return _text(reply, 'prompt');
  }

  /// One reference image from [prompt], with the painter's saved defaults.
  Future<PersonaImage> generateImage(
    ModelConnection painter,
    String prompt, {
    GenerationCancellation? cancellation,
  }) async {
    final factory = runner.imageProviderFactory, library = runner.assetLibrary;
    if (factory == null || library == null) {
      throw StateError('生图服务或本地图片目录尚未配置');
    }
    final illustrations = IllustrationService(
      repository: repository,
      runner: runner,
    );
    final frozen = await illustrations.withLatestWorkflow(painter);
    final request = IllustrationService.requestFrom(
      frozen,
      prompt,
      await illustrations.savedDefaults(painter),
    );
    final provider = await factory(frozen);
    try {
      final output = await provider.generate(
        request,
        cancellation: cancellation,
      );
      return PersonaImage.of(await library.prepare(output.bytes));
    } finally {
      provider.close();
    }
  }

  Future<Map<String, dynamic>> _ask(
    ModelConnection connection,
    String instructions,
    Object content,
    GenerationCancellation? cancellation,
  ) async {
    final provider = await runner.providerFactory(connection);
    try {
      final text = StringBuffer();
      await for (final event in provider.generate(
        ProviderRequest(
          messages: [
            {'role': 'system', 'content': instructions},
            {'role': 'user', 'content': content},
          ],
        ),
        cancellation: cancellation,
      )) {
        if (event is TextDelta) text.write(event.text);
      }
      final cleaned = text
          .toString()
          .trim()
          .replaceFirst(RegExp(r'^```(?:json)?\s*'), '')
          .replaceFirst(RegExp(r'\s*```$'), '');
      try {
        return jsonObject(jsonDecode(cleaned), 'reply');
      } on FormatException {
        throw const FormatException('模型没有按要求返回 JSON，请重试');
      }
    } finally {
      provider.close();
    }
  }

  static String _text(Map<String, dynamic> reply, String key) {
    final value = reply[key];
    if (value is! String) throw FormatException('模型回复缺少 $key');
    return value.trim();
  }
}

```dart
  Persona? _undo;

  /// Model results replace the text fields; one step back restores them.
  void _apply(Persona result) {
    if (!mounted) return;
    setState(() {
      _undo = _draft;
      _fill(result);
    });
  }

  Future<ModelConnection> _connection(String role) async =>
      await ref.read(servicesProvider).connectionFor(role) ??
      (throw FormatException(
        role == 'image' ? '请先在设置中配置插画模型' : '请先在设置中配置资料整理模型',
      ));

  Future<void> _describe() => _run(() async {
    _apply(
      await ref
          .read(servicesProvider)
          .personas
          .describeImages(
            await _connection('analysis'),
            _draft,
            cancellation: _cancel,
          ),
    );
  });

  Future<void> _refine() => _run(() async {
    _apply(
      await ref
          .read(servicesProvider)
          .personas
          .refine(await _connection('analysis'), _draft, cancellation: _cancel),
    );
  });

  Future<void> _generateImage() => _run(() async {
    _checkRoom();
    final personas = ref.read(servicesProvider).personas;
    final painter = await _connection('image');
    final drafted = await personas.imagePrompt(
      await _connection('analysis'),
      _draft,
      painter,
      cancellation: _cancel,

    final prompt = await askText(
      context,
      title: '参考图提示词',
      initial: drafted,
      label: '提示词',
      maxLines: 6,
    );
    if (prompt == null) return;
    final image = await personas.generateImage(
      painter,
      prompt,
      cancellation: _cancel,
    );
    if (mounted) setState(() => _images = [..._images, image]);
```
